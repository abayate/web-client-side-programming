// Lab 5 - Task 3: image generation modality.
// Based on the "image generation" samples in https://github.com/openai/openai-node/tree/master/examples
// The API returns base64 PNG data, so we decode it and write it to disk ourselves.

import OpenAI from 'openai';
import dotenv from 'dotenv';
import fs from 'node:fs/promises';
import path from 'node:path';

// override: true so a stale machine-level OPENAI_API_KEY cannot shadow our .env file.
dotenv.config({ override: true });

const openai = new OpenAI();

const OUTPUT_DIR = 'output';
const PROMPT =
  'A clean, flat vector illustration of a laptop on a desk. The laptop screen shows a ' +
  'simple neural network diagram made of glowing nodes and connecting lines. ' +
  'Soft indigo and violet color palette, plenty of white space, no text anywhere in the image.';

async function main() {
  await fs.mkdir(OUTPUT_DIR, { recursive: true });

  console.log('--- Image Generation (gpt-image-1-mini) ---');
  console.log(`Prompt: ${PROMPT}\n`);
  console.log('Generating... (this usually takes 10-30 seconds)');

  const started = Date.now();
  const response = await openai.images.generate({
    model: 'gpt-image-1-mini',
    prompt: PROMPT,
    size: '1024x1024',
    n: 1,
  });

  const b64 = response.data[0].b64_json;
  if (!b64) throw new Error('The API did not return base64 image data.');

  const outPath = path.join(OUTPUT_DIR, 'generated-image.png');
  await fs.writeFile(outPath, Buffer.from(b64, 'base64'));

  const bytes = Buffer.byteLength(b64, 'base64');
  console.log(`\nDone in ${((Date.now() - started) / 1000).toFixed(1)}s`);
  console.log(`Saved: ${outPath} (${(bytes / 1024).toFixed(1)} KB)`);
  if (response.usage) {
    console.log(`Tokens: ${JSON.stringify(response.usage)}`);
  }
}

main().catch((error) => {
  console.error('Image generation failed:', error.message);
  process.exit(1);
});

// Lab 5 - Task 3: speech (text-to-speech) modality.
// Based on the "speech" samples in https://github.com/openai/openai-node/tree/master/examples
// The response is a web Response object, so we pull an ArrayBuffer out of it and write an MP3.

import OpenAI from 'openai';
import dotenv from 'dotenv';
import fs from 'node:fs/promises';
import path from 'node:path';

dotenv.config({ override: true });

const openai = new OpenAI();

const OUTPUT_DIR = 'output';
const SCRIPT_TEXT =
  'Welcome to Lab Five. In this lab I built four things: a TensorFlow dot J S image ' +
  'classifier that runs entirely in the browser, a multi model comparison script, ' +
  'a LangChain guardian pipeline, and a retrieval augmented resume analyzer. ' +
  'This audio itself was generated by the OpenAI text to speech endpoint.';

// Each voice has a different character; generating two makes the difference audible.
const VOICES = ['nova', 'onyx'];

async function speak(voice) {
  console.log(`Synthesizing with voice "${voice}"...`);
  const started = Date.now();

  const response = await openai.audio.speech.create({
    model: 'gpt-4o-mini-tts',
    voice,
    input: SCRIPT_TEXT,
    instructions: 'Speak clearly and at a steady, confident pace, like a student presenting a lab demo.',
  });

  const buffer = Buffer.from(await response.arrayBuffer());
  const outPath = path.join(OUTPUT_DIR, `lab5-narration-${voice}.mp3`);
  await fs.writeFile(outPath, buffer);

  console.log(
    `  Saved ${outPath} (${(buffer.length / 1024).toFixed(1)} KB) in ${((Date.now() - started) / 1000).toFixed(1)}s`
  );
}

async function main() {
  await fs.mkdir(OUTPUT_DIR, { recursive: true });

  console.log('--- Text to Speech (gpt-4o-mini-tts) ---');
  console.log(`Script (${SCRIPT_TEXT.length} characters):\n${SCRIPT_TEXT}\n`);

  for (const voice of VOICES) {
    await speak(voice);
  }

  console.log('\nOpen the .mp3 files in the output/ folder to hear the result.');
}

main().catch((error) => {
  console.error('Speech generation failed:', error.message);
  process.exit(1);
});

// Lab 5 - Task 3: vision (image input) modality.
// This is the interesting comparison for this lab: in Task 1, MobileNet running in the
// browser labelled my photo "Labrador retriever, 64.63%" - a single ImageNet class and
// nothing else. Here the SAME photo goes to a multimodal LLM, which can actually describe
// the scene and answer follow-up questions about it. Same input, very different capability.

import OpenAI from 'openai';
import dotenv from 'dotenv';
import fs from 'node:fs/promises';

dotenv.config({ override: true });

const openai = new OpenAI();

// The exact image Task 1 classified with MobileNet.
const IMAGE_PATH = '../task1_tensorflowjs/img/labrador.jpg';

async function main() {
  console.log('--- Vision / Image Understanding (gpt-4o) ---');
  console.log(`Image: ${IMAGE_PATH}`);
  console.log('MobileNet (Task 1, in-browser) said: "Labrador retriever" at 64.63% confidence.\n');

  const imageBuffer = await fs.readFile(IMAGE_PATH);
  const dataUrl = `data:image/jpeg;base64,${imageBuffer.toString('base64')}`;

  const completion = await openai.chat.completions.create({
    model: 'gpt-4o',
    messages: [
      {
        role: 'user',
        content: [
          {
            type: 'text',
            text:
              'A MobileNet model running in a browser labelled this image "Labrador retriever" ' +
              'with 64.63% confidence. Do you agree with that label? Then describe everything ' +
              'else that is happening in this picture that a single-label classifier would ' +
              'completely miss. Answer in under 150 words.',
          },
          { type: 'image_url', image_url: { url: dataUrl, detail: 'low' } },
        ],
      },
    ],
  });

  console.log('GPT-4o response:');
  console.log(completion.choices[0]?.message?.content);
  console.log(`\nTokens used: ${JSON.stringify(completion.usage)}`);
}

main().catch((error) => {
  console.error('Vision request failed:', error.message);
  process.exit(1);
});
