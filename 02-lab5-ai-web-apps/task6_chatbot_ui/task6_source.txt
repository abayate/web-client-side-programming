// Lab 5 - Task 6: Express backend for the resume-analyzer chatbot UI.
// Anthony Bayate - TECH*3500*01
//
// Task 6 asks for the Task 5 RAG code with a front end on top: the PDF is uploaded from
// the GUI and analyzed by the backend. The split is:
//   public/     - the browser UI (no API key ever reaches the client)
//   rag.js      - the LangChain pipeline from Task 5
//   server.js   - this file: upload endpoint, analysis endpoint, chat endpoint

import express from "express";
import multer from "multer";
import path from "node:path";
import { randomUUID } from "node:crypto";
import { fileURLToPath } from "node:url";
import dotenv from "dotenv";

import { buildResumeIndex, analyzeResume, askFollowUp } from "./rag.js";

// override: true so a stale machine-level OPENAI_API_KEY cannot shadow .env
dotenv.config({ override: true });

const __dirname = path.dirname(fileURLToPath(import.meta.url));

const app = express();
const PORT = process.env.PORT || 3000;

// Sessions live in memory only: uploaded resumes are never written to disk.
// Each entry is { retriever, history, ... } plus a createdAt for expiry.
const sessions = new Map();
const SESSION_TTL_MS = 60 * 60 * 1000; // 1 hour

function pruneSessions() {
  const cutoff = Date.now() - SESSION_TTL_MS;
  for (const [id, session] of sessions) {
    if (session.createdAt < cutoff) sessions.delete(id);
  }
}
setInterval(pruneSessions, 10 * 60 * 1000).unref();

// Keep the PDF in memory rather than writing it to an uploads/ folder.
const upload = multer({
  storage: multer.memoryStorage(),
  limits: { fileSize: 10 * 1024 * 1024 }, // 10 MB
  fileFilter: (req, file, cb) => {
    if (file.mimetype === "application/pdf") return cb(null, true);
    cb(new Error("Only PDF files are accepted."));
  },
});

app.use(express.json({ limit: "1mb" }));
app.use(express.static(path.join(__dirname, "public")));

// --- POST /api/analyze --------------------------------------------------------
// Receives the PDF, builds the vector index, runs the four-chain analysis, and
// returns a session id the client uses for follow-up chat.
app.post("/api/analyze", upload.single("resume"), async (req, res) => {
  const startedAt = Date.now();
  try {
    if (!req.file) {
      return res.status(400).json({ error: "No file was uploaded." });
    }

    const role = (req.body.role || "a mid-level front-end engineer").slice(0, 120);

    console.log(`[analyze] ${req.file.originalname} (${(req.file.size / 1024).toFixed(0)} KB), role="${role}"`);

    const session = await buildResumeIndex(req.file.buffer);
    session.createdAt = Date.now();
    session.filename = req.file.originalname;

    const analysis = await analyzeResume(session, role);

    const sessionId = randomUUID();
    sessions.set(sessionId, session);

    const elapsedMs = Date.now() - startedAt;
    console.log(`[analyze] done in ${(elapsedMs / 1000).toFixed(1)}s -> session ${sessionId}`);

    res.json({
      sessionId,
      meta: {
        filename: session.filename,
        pageCount: session.pageCount,
        chunkCount: session.chunkCount,
        charCount: session.charCount,
        k: session.k,
        elapsedMs,
        role,
      },
      ...analysis,
    });
  } catch (error) {
    console.error("[analyze] failed:", error.message);
    res.status(500).json({ error: error.message || "Analysis failed." });
  }
});

// --- POST /api/chat -----------------------------------------------------------
// A follow-up question about a resume that was already uploaded.
app.post("/api/chat", async (req, res) => {
  try {
    const { sessionId, question } = req.body || {};

    if (!sessionId || !sessions.has(sessionId)) {
      return res
        .status(404)
        .json({ error: "That resume session expired. Please upload the PDF again." });
    }
    if (!question || !question.trim()) {
      return res.status(400).json({ error: "Question cannot be empty." });
    }

    const session = sessions.get(sessionId);
    const answer = await askFollowUp(session, question.trim().slice(0, 1000));
    res.json({ answer });
  } catch (error) {
    console.error("[chat] failed:", error.message);
    res.status(500).json({ error: error.message || "Chat request failed." });
  }
});

app.get("/api/health", (req, res) => {
  res.json({ ok: true, activeSessions: sessions.size });
});

// Multer and other middleware errors arrive here (e.g. file too large, wrong type).
app.use((error, req, res, next) => {
  console.error("[error]", error.message);
  const status = error.code === "LIMIT_FILE_SIZE" ? 413 : 400;
  res.status(status).json({ error: error.message });
});

app.listen(PORT, () => {
  console.log(`Resume Analyzer running at http://localhost:${PORT}`);
  if (!process.env.OPENAI_API_KEY) {
    console.warn("WARNING: OPENAI_API_KEY is not set. Put it in a .env file next to server.js.");
  }
});

// Lab 5 - Task 6: the RAG engine, extracted from Task 5 so the web server can reuse it.
// Anthony Bayate - TECH*3500*01
//
// Task 5 was a terminal script that read a path from readline. For Task 6 the PDF
// arrives from the browser as an in-memory Buffer, so the only real change here is that
// every function takes a Buffer instead of a file path. All the tuning from Task 5
// (500/100 chunks, adaptive-k MMR retrieval, grounded prompts, chain-of-chains) is kept.

import { ChatOpenAI, OpenAIEmbeddings } from "@langchain/openai";
import { PromptTemplate } from "@langchain/core/prompts";
import { RunnableSequence } from "@langchain/core/runnables";
import { StringOutputParser } from "@langchain/core/output_parsers";
import { formatDocumentsAsString } from "langchain/util/document";
import { MemoryVectorStore } from "langchain/vectorstores/memory";
import { RecursiveCharacterTextSplitter } from "langchain/text_splitter";

import { createRequire } from "module";
const require = createRequire(import.meta.url);
// Requiring "pdf-parse" directly triggers its debug mode under ESM and crashes with
// ENOENT on a test fixture that is not shipped. Requiring the lib file skips that.
const pdf = require("pdf-parse/lib/pdf-parse.js");

const GROUNDING_RULES = `
Rules you must follow:
- Use ONLY the resume context below. Do not use outside knowledge about the company or person.
- Every claim you make must be traceable to the context. Quote the exact phrase you are relying on in "quotes".
- If the context does not cover something, write "not stated in the resume" instead of guessing.
`;

const praisingTemplate = `You are an enthusiastic hiring manager trying to convince your team to hire a candidate.
Based ONLY on the following resume context, write a compelling summary praising the candidate.
Focus on their strengths, achievements, and skills. Prefer concrete, measurable results over adjectives.
Do not mention any weaknesses or gaps.
${GROUNDING_RULES}
Context:
{context}

Question:
{question}

Praising Summary:`;

const criticalTemplate = `You are a cautious and objective technical recruiter.
Based ONLY on the following resume context, identify potential weaknesses, gaps, or areas that need clarification.
Be fair and objective. Do not invent information. If there are no clear weaknesses, state that the resume looks strong.
For each point, say explicitly whether it is a real gap in the resume or simply something the resume does not mention.
${GROUNDING_RULES}
Context:
{context}

Question:
{question}

Critical Analysis:`;

const extractionTemplate = `You are a resume parser. Extract structured facts from the resume context below.
Respond with ONLY a valid JSON object and no markdown fences, using exactly this shape:
{{
  "name": string | null,
  "years_of_experience": number | null,
  "current_title": string | null,
  "top_skills": string[],
  "employers": string[],
  "education_status": string | null,
  "quantified_achievements": string[]
}}
Use null or an empty array for anything the resume does not state. Do not invent values.

Context:
{context}

Question:
{question}

JSON:`;

const recommendationTemplate = `You are the hiring committee chair. Two of your colleagues reviewed the same candidate.

The advocate wrote:
{praise}

The skeptic wrote:
{critique}

Weigh both views against each other and produce a final decision for the role: {role}.
Respond in exactly this format:

VERDICT: <ADVANCE TO INTERVIEW | ADVANCE WITH RESERVATIONS | DO NOT ADVANCE>
CONFIDENCE: <High | Medium | Low>
REASONING: <2-3 sentences that explicitly weigh the strongest point from each reviewer>
QUESTIONS TO ASK: <exactly 3 numbered interview questions targeting the skeptic's concerns>`;

// The follow-up chat prompt. This is what makes Task 6 a chatbot rather than a
// one-shot report: the user keeps asking questions and each one is answered by
// retrieving from the same vector store built at upload time.
const chatTemplate = `You are a recruiting assistant answering questions about ONE specific resume.

Resume context retrieved for this question:
{context}

Conversation so far:
{history}

Answer the recruiter's question using ONLY the resume context above.
If the resume does not contain the answer, say so plainly - do not speculate.
Keep the answer under 120 words unless the question requires a list.

Recruiter's question: {question}

Answer:`;

/**
 * Parse a PDF buffer, chunk it, embed it, and return a reusable session object.
 * The vector store is built once per upload and then reused by both the initial
 * analysis and every follow-up chat message.
 */
export async function buildResumeIndex(pdfBuffer) {
  const pdfData = await pdf(pdfBuffer);
  const resumeText = pdfData.text;

  if (!resumeText || !resumeText.trim()) {
    throw new Error(
      "Could not extract text from this PDF. It may be a scanned/image-only document."
    );
  }

  const textSplitter = new RecursiveCharacterTextSplitter({
    chunkSize: 500,
    chunkOverlap: 100,
    separators: ["\n\n", "\n", ". ", " ", ""],
  });
  const docs = await textSplitter.createDocuments([resumeText]);

  const embeddings = new OpenAIEmbeddings({ model: "text-embedding-3-small" });
  const vectorStore = await MemoryVectorStore.fromDocuments(docs, embeddings);

  // k scales with the document so a short resume never loses a whole section.
  const k = Math.min(8, docs.length);
  const retriever = vectorStore.asRetriever({
    k,
    searchType: "mmr",
    searchKwargs: { fetchK: Math.min(20, docs.length * 2), lambda: 0.6 },
  });

  return {
    retriever,
    chunkCount: docs.length,
    pageCount: pdfData.numpages,
    charCount: resumeText.length,
    k,
    history: [], // follow-up chat turns, appended as { role, content }
  };
}

function createRagChain(retriever, template, llm) {
  return RunnableSequence.from([
    {
      context: RunnableSequence.from([
        (input) => input.question,
        retriever,
        formatDocumentsAsString,
      ]),
      question: (input) => input.question,
    },
    PromptTemplate.fromTemplate(template),
    llm,
    new StringOutputParser(),
  ]);
}

/** Run the four-chain analysis over an already-built session. */
export async function analyzeResume(session, role = "a mid-level front-end engineer") {
  const llm = new ChatOpenAI({ model: "gpt-4o", temperature: 0.1 });

  const praisingChain = createRagChain(session.retriever, praisingTemplate, llm);
  const criticalChain = createRagChain(session.retriever, criticalTemplate, llm);
  const extractionChain = createRagChain(session.retriever, extractionTemplate, llm);

  // Independent chains, so run them concurrently.
  const [praise, critique, extractionRaw] = await Promise.all([
    praisingChain.invoke({
      question:
        "What are this candidate's strongest achievements, measurable results, and technical skills?",
    }),
    criticalChain.invoke({
      question:
        "What gaps, missing skills, short tenures, or unclear claims should a recruiter ask about?",
    }),
    extractionChain.invoke({
      question: "Extract the candidate's name, experience, skills, employers, and education.",
    }),
  ]);

  // Chain-of-chains: this call consumes the output of the two above.
  const recommendationChain = RunnableSequence.from([
    PromptTemplate.fromTemplate(recommendationTemplate),
    new ChatOpenAI({ model: "gpt-4o", temperature: 0 }),
    new StringOutputParser(),
  ]);
  const recommendation = await recommendationChain.invoke({ praise, critique, role });

  let facts = null;
  try {
    facts = JSON.parse(extractionRaw.replace(/^```(?:json)?\s*|\s*```$/g, "").trim());
  } catch {
    facts = null; // the UI falls back to hiding the facts panel
  }

  return { praise, critique, facts, recommendation };
}

/** Answer one follow-up question against the uploaded resume. */
export async function askFollowUp(session, question) {
  const llm = new ChatOpenAI({ model: "gpt-4o", temperature: 0.2 });

  // Only the last few turns are sent back, to keep the prompt small.
  const history =
    session.history
      .slice(-6)
      .map((m) => `${m.role === "user" ? "Recruiter" : "Assistant"}: ${m.content}`)
      .join("\n") || "(this is the first question)";

  const chain = RunnableSequence.from([
    {
      context: RunnableSequence.from([
        (input) => input.question,
        session.retriever,
        formatDocumentsAsString,
      ]),
      question: (input) => input.question,
      history: (input) => input.history,
    },
    PromptTemplate.fromTemplate(chatTemplate),
    llm,
    new StringOutputParser(),
  ]);

  const answer = await chain.invoke({ question, history });

  session.history.push({ role: "user", content: question });
  session.history.push({ role: "assistant", content: answer });

  return answer;
}

// Lab 5 - Task 6: front-end logic for the resume analyzer.
// Talks to the Node backend over fetch. No API key is ever present in the browser.

const uploadView = document.getElementById('upload-view');
const loadingView = document.getElementById('loading-view');
const resultsView = document.getElementById('results-view');

const dropzone = document.getElementById('dropzone');
const fileInput = document.getElementById('file-input');
const fileChosen = document.getElementById('file-chosen');
const roleInput = document.getElementById('role-input');
const uploadForm = document.getElementById('upload-form');
const analyzeBtn = document.getElementById('analyze-btn');
const uploadError = document.getElementById('upload-error');
const newUploadBtn = document.getElementById('new-upload-btn');

const chatForm = document.getElementById('chat-form');
const chatInput = document.getElementById('chat-input');
const chatSend = document.getElementById('chat-send');
const chatLog = document.getElementById('chat-log');
const suggestions = document.getElementById('suggestions');

let selectedFile = null;
let sessionId = null;

// --- view switching ---

function showView(view) {
    uploadView.hidden = view !== 'upload';
    loadingView.hidden = view !== 'loading';
    resultsView.hidden = view !== 'results';
}

function showError(message) {
    uploadError.textContent = message;
    uploadError.hidden = false;
}

function clearError() {
    uploadError.hidden = true;
}

// --- file selection ---

function acceptFile(file) {
    if (!file) return;

    if (file.type !== 'application/pdf') {
        showError('Please choose a PDF file. Word documents and images are not supported.');
        return;
    }
    if (file.size > 10 * 1024 * 1024) {
        showError('That file is larger than the 10 MB limit.');
        return;
    }

    clearError();
    selectedFile = file;
    fileChosen.textContent = `Selected: ${file.name} (${(file.size / 1024).toFixed(0)} KB)`;
    fileChosen.hidden = false;
    analyzeBtn.disabled = false;
}

dropzone.addEventListener('click', () => fileInput.click());
dropzone.addEventListener('keydown', (event) => {
    if (event.key === 'Enter' || event.key === ' ') {
        event.preventDefault();
        fileInput.click();
    }
});

fileInput.addEventListener('change', (event) => acceptFile(event.target.files[0]));

['dragenter', 'dragover'].forEach((type) => {
    dropzone.addEventListener(type, (event) => {
        event.preventDefault();
        dropzone.classList.add('dragover');
    });
});

['dragleave', 'drop'].forEach((type) => {
    dropzone.addEventListener(type, (event) => {
        event.preventDefault();
        dropzone.classList.remove('dragover');
    });
});

dropzone.addEventListener('drop', (event) => {
    acceptFile(event.dataTransfer.files[0]);
});

// --- fake-but-honest progress ticker for the loading view ---
// The backend does not stream progress, so these advance on a timer that roughly
// matches how long each stage actually takes.
let stepTimers = [];

function runLoadingSteps() {
    const steps = [...loadingView.querySelectorAll('.steps li')];
    steps.forEach((li) => li.classList.remove('active', 'done'));
    stepTimers.forEach(clearTimeout);
    stepTimers = [];

    const delays = [0, 1200, 3200, 9000];
    steps.forEach((li, index) => {
        stepTimers.push(setTimeout(() => {
            steps.slice(0, index).forEach((prev) => {
                prev.classList.remove('active');
                prev.classList.add('done');
            });
            li.classList.add('active');
        }, delays[index]));
    });
}

// --- submit ---

uploadForm.addEventListener('submit', async (event) => {
    event.preventDefault();
    if (!selectedFile) return;

    clearError();
    showView('loading');
    runLoadingSteps();

    const formData = new FormData();
    formData.append('resume', selectedFile);
    formData.append('role', roleInput.value.trim() || 'a mid-level front-end engineer');

    try {
        const response = await fetch('/api/analyze', { method: 'POST', body: formData });
        const data = await response.json();

        if (!response.ok) throw new Error(data.error || `Server returned ${response.status}`);

        sessionId = data.sessionId;
        renderResults(data);
        showView('results');
    } catch (error) {
        stepTimers.forEach(clearTimeout);
        showView('upload');
        showError(error.message);
    }
});

newUploadBtn.addEventListener('click', () => {

    sessionId = null;
    fileInput.value = '';
    fileChosen.hidden = true;
    analyzeBtn.disabled = true;
    chatLog.innerHTML = '';
    suggestions.hidden = false;
    clearError();
    showView('upload');
});

// --- rendering ---

// Minimal markdown: **bold** plus paragraph breaks. Everything is inserted as text
// nodes or through a whitelist, so model output can never inject HTML.
function renderProse(container, text) {
    container.innerHTML = '';
    text.split(/\n{2,}/).forEach((block) => {
        const p = document.createElement('p');
        const parts = block.split(/\*\*(.+?)\*\*/g);
        parts.forEach((part, index) => {
            if (index % 2 === 1) {
                const strong = document.createElement('strong');
                strong.textContent = part;
                p.appendChild(strong);
            } else if (part) {
                p.appendChild(document.createTextNode(part));
            }
        });
        container.appendChild(p);
    });
}

function renderFacts(facts) {
    const dl = document.getElementById('out-facts');
    const card = document.getElementById('facts-card');
    dl.innerHTML = '';

    if (!facts) {
        card.hidden = true;
        return;
    }
    card.hidden = false;

    const rows = [
        ['Name', facts.name],
        ['Current title', facts.current_title],
        ['Years of experience', facts.years_of_experience],
        ['Education', facts.education_status],
        ['Top skills', facts.top_skills],
        ['Employers', facts.employers],
        ['Quantified achievements', facts.quantified_achievements],
    ];

    for (const [label, value] of rows) {
        const isEmptyArray = Array.isArray(value) && value.length === 0;
        if (value === null || value === undefined || value === '' || isEmptyArray) continue;

        const dt = document.createElement('dt');
        dt.textContent = label;
        const dd = document.createElement('dd');

        if (Array.isArray(value) && label === 'Quantified achievements') {
            const ul = document.createElement('ul');
            value.forEach((item) => {
                const li = document.createElement('li');
                li.textContent = item;
                ul.appendChild(li);
            });
            dd.appendChild(ul);
        } else if (Array.isArray(value)) {
            const row = document.createElement('div');
            row.className = 'pill-row';
            value.forEach((item) => {
                const span = document.createElement('span');
                span.className = 'pill';
                span.textContent = item;
                row.appendChild(span);
            });
            dd.appendChild(row);
        } else {
            dd.textContent = String(value);
        }

        dl.append(dt, dd);
    }
}

function renderResults(data) {
    document.getElementById('result-filename').textContent = data.meta.filename;
    document.getElementById('result-meta').textContent =
        `${data.meta.pageCount} page(s) · ${data.meta.charCount.toLocaleString()} characters · ` +
        `${data.meta.chunkCount} chunks · retriever k=${data.meta.k} · ` +
        `analyzed in ${(data.meta.elapsedMs / 1000).toFixed(1)}s · role: ${data.meta.role}`;

    document.getElementById('out-recommendation').textContent = data.recommendation;
    renderProse(document.getElementById('out-praise'), data.praise);
    renderProse(document.getElementById('out-critique'), data.critique);
    renderFacts(data.facts);
}

// --- chat ---

function addMessage(text, variant) {
    const div = document.createElement('div');
    div.className = `msg msg-${variant}`;
    div.textContent = text;
    chatLog.appendChild(div);
    chatLog.scrollTop = chatLog.scrollHeight;
    return div;
}

async function ask(question) {
    if (!question || !sessionId) return;

    suggestions.hidden = true;
    addMessage(question, 'user');
    chatInput.value = '';
    chatInput.disabled = true;
    chatSend.disabled = true;

    const typing = addMessage('Thinking…', 'bot msg-typing');

    try {
        const response = await fetch('/api/chat', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ sessionId, question }),
        });
        const data = await response.json();

        if (!response.ok) throw new Error(data.error || `Server returned ${response.status}`);

        typing.classList.remove('msg-typing');
        typing.textContent = data.answer;
    } catch (error) {
        typing.classList.remove('msg-typing');
        typing.classList.add('msg-error');
        typing.textContent = error.message;
    } finally {
        chatInput.disabled = false;
        chatSend.disabled = false;
        chatInput.focus();
        chatLog.scrollTop = chatLog.scrollHeight;
    }
}

chatForm.addEventListener('submit', (event) => {
    event.preventDefault();
    ask(chatInput.value.trim());
});

suggestions.addEventListener('click', (event) => {
    if (event.target.classList.contains('chip')) {
        ask(event.target.textContent.trim());
    }
});
